import pygame import random import sys import time # --- AYARLAR VE BAŞLATMA --- pygame.init() WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Galaktik Savaşçı: Power-up Devri") clock = pygame.time.Clock() # --- RENKLER --- WHITE = (255, 255, 255) BLACK = (0, 0, 0) RED = (255, 50, 50) BLUE = (50, 150, 255) YELLOW = (255, 255, 0) PURPLE = (160, 32, 240) CYAN = (0, 255, 255) # --- FONT ARŞİVİ --- font_small = pygame.font.SysFont("Courier New", 20, bold=True) font_med = pygame.font.SysFont("Courier New", 35, bold=True) font_large = pygame.font.SysFont("Courier New", 60, bold=True) # --- SINIFLAR --- class Player: def __init__(self): self.size = 50 self.x = WIDTH // 2 self.y = HEIGHT - 80 self.speed = 9 self.triple_shot_active = False self.powerup_timer = 0 def draw(self): # Güçlendirici aktifse uçağın rengi değişsin color = CYAN if self.triple_shot_active else BLUE # Ana gövde pygame.draw.polygon(screen, color, [(self.x, self.y + 30), (self.x + 25, self.y), (self.x + 50, self.y + 30)]) pygame.draw.rect(screen, WHITE, (self.x + 20, self.y + 30, 10, 15)) # Motor if self.triple_shot_active: # Kalkan veya güç efekti dairesi pygame.draw.circle(screen, CYAN, (self.x + 25, self.y + 20), 35, 2) def move(self, keys): if keys[pygame.K_LEFT] and self.x > 0: self.x -= self.speed if keys[pygame.K_RIGHT] and self.x < WIDTH - self.size: self.x += self.speed class Bullet: def __init__(self, x, y, angle=0): self.x = x self.y = y self.speed_y = -12 self.speed_x = angle # Yanlara ateş için def update(self): self.y += self.speed_y self.x += self.speed_x def draw(self): pygame.draw.rect(screen, YELLOW, (self.x, self.y, 6, 15)) class Enemy: def __init__(self, level): self.size = 45 self.x = random.randint(0, WIDTH - self.size) self.y = random.randint(-150, -50) self.speed = random.uniform(2, 4) + (level * 0.4) def update(self): self.y += self.speed def draw(self): pygame.draw.rect(screen, RED, (self.x, self.y, self.size, self.size)) pygame.draw.line(screen, WHITE, (self.x, self.y), (self.x + self.size, self.y + self.size), 2) class PowerUp: def __init__(self): self.size = 30 self.x = random.randint(0, WIDTH - self.size) self.y = -50 self.speed = 4 def update(self): self.y += self.speed def draw(self): pygame.draw.circle(screen, CYAN, (self.x + 15, self.y + 15), 15) text = font_small.render("3", True, BLACK) screen.blit(text, (self.x + 10, self.y + 5)) # --- YARDIMCI FONKSİYONLAR --- def message_display(text, font, color, y_offset=0): text_surface = font.render(text, True, color) text_rect = text_surface.get_rect(center=(WIDTH/2, HEIGHT/2 + y_offset)) screen.blit(text_surface, text_rect) # --- ANA OYUN FONKSİYONU --- def main_game(): player = Player() bullets = [] enemies = [] powerups = [] score = 0 lives = 3 level = 1 running = True # Zamanlayıcılar enemy_spawn_event = pygame.USEREVENT + 1 pygame.time.set_timer(enemy_spawn_event, 1000) powerup_spawn_event = pygame.USEREVENT + 2 pygame.time.set_timer(powerup_spawn_event, 10000) # 10 saniyede bir powerup while running: screen.fill(BLACK) # 1. Event Handling for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit(); sys.exit() if event.type == enemy_spawn_event: enemies.append(Enemy(level)) if event.type == powerup_spawn_event: powerups.append(PowerUp()) if event.type == pygame.KEYDOWN: if event.key == pygame.K_SPACE: if player.triple_shot_active: bullets.append(Bullet(player.x + 22, player.y, 0)) bullets.append(Bullet(player.x + 5, player.y, -3)) bullets.append(Bullet(player.x + 40, player.y, 3)) else: bullets.append(Bullet(player.x + 22, player.y)) # 2. Update Logic keys = pygame.key.get_pressed() player.move(keys) # Power-up Süre Kontrolü if player.triple_shot_active: if time.time() > player.powerup_timer: player.triple_shot_active = False # Mermileri Güncelle for b in bullets[:]: b.update() if b.y < 0: bullets.remove(b) # Düşmanları Güncelle for e in enemies[:]: e.update() if e.y > HEIGHT: enemies.remove(e) lives -= 1 # Mermi-Düşman Çarpışması for b in bullets[:]: if (e.x < b.x < e.x + e.size) and (e.y < b.y < e.y + e.size): score += 15 if b in bullets: bullets.remove(b) if e in enemies: enemies.remove(e) break # Power-up Toplama for p in powerups[:]: p.update() if (player.x < p.x + p.size and player.x + player.size > p.x and player.y < p.y + p.size and player.y + player.size > p.y): player.triple_shot_active = True player.powerup_timer = time.time() + 7 # 7 saniye sürer powerups.remove(p) elif p.y > HEIGHT: powerups.remove(p) # Zorluk Seviyesi level = 1 + (score // 200) # 3. Drawing player.draw() for b in bullets: b.draw() for e in enemies: e.draw() for p in powerups: p.draw() # UI show_info = f"SKOR: {score} CAN: {lives} SEVİYE: {level}" screen.blit(font_small.render(show_info, True, WHITE), (10, 10)) if player.triple_shot_active: rem = int(player.powerup_timer - time.time()) screen.blit(font_small.render(f"GÜÇ: {rem}s", True, CYAN), (10, 40)) if lives <= 0: running = False pygame.display.flip() clock.tick(60) return score # --- ANA DÖNGÜ --- while True: screen.fill(BLACK) message_display("GALAKTİK SAVAŞÇI", font_large, BLUE, -50) message_display("BAŞLAMAK İÇİN 'B' TUŞUNA BASIN", font_small, WHITE, 50) pygame.display.flip() waiting = True while waiting: for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit(); sys.exit() if event.type == pygame.KEYDOWN and event.key == pygame.K_b: waiting = False final_score = main_game() screen.fill(BLACK) message_display("OYUN BİTTİ", font_large, RED, -50) message_display(f"SKORUNUZ: {final_score}", font_med, WHITE, 50) message_display("TEKRAR İÇİN 'T' TUŞUNA BASIN", font_small, YELLOW, 120) pygame.display.flip() retry = True while retry: for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit(); sys.exit() if event.type == pygame.KEYDOWN and event.key == pygame.K_t: retry = False
